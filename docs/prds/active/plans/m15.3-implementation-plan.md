# M15.3: Grocery Lists UX

## Context

M15.3 transforms the grocery list experience from flat lists with linear progress bars into card-based layouts with circular progress rings, animated check-offs with haptics, collapsible category sections, a sticky bottom progress bar, and a 100% completion celebration. This is the most user-facing phase — grocery shopping is the primary daily workflow.

**Depends on**: M15.1 (ForagerTheme tokens, Liquid Glass TabView) + M15.2 (all colors/fonts migrated to tokens).

**What changes**: WeeklyListsView layout (flat → cards), GroceryListDetailView layout (progress header → sticky bottom, quick-add → bottom, collapsible sections, haptics), creation flow (single action → 3-option action sheet), new shared components (ForagerCard, ForagerProgressRing, ForagerSectionHeader, CategoryChipPills).

**What does NOT change**: Data model, Core Data queries, service layer, template/autocomplete system.

struct ForagerProgressRing: View {
    let progress: Double  // 0.0–1.0
    var size: CGFloat = 56

    var body: some View {
        ZStack {
            // Background track
            Circle()
                .stroke(ForagerTheme.backgroundTertiary, lineWidth: 4)

            // Progress arc
            Circle()
                .trim(from: 0, to: progress)
                .stroke(progressColor, style: StrokeStyle(lineWidth: 4, lineCap: .round))
                .rotationEffect(.degrees(-90))
                .animation(.easeInOut(duration: 0.3), value: progress)

            // Percentage text
            Text("\(Int(progress * 100))%")
                .font(ForagerTheme.captionFont)
                .foregroundStyle(ForagerTheme.textSecondary)
        }
        .frame(width: size, height: size)
        .accessibilityLabel("List progress")
        .accessibilityValue("\(Int(progress * 100)) percent complete")
    }

    private var progressColor: Color {
        if progress >= 1.0 { return ForagerTheme.statusSuccessFG }  // springGreen
        if progress >= 0.5 { return ForagerTheme.accentSecondary }   // leafGreen
        return ForagerTheme.accentPrimary                            // forestGreen
    }
}
```

```swift
struct WeeklyListRowView: View {
    @ObservedObject var weeklyList: WeeklyList
    @FetchRequest private var itemsFetch: FetchedResults<GroceryListItem>

    var body: some View {
        VStack(spacing: 0) {
            // Main card body
            HStack(alignment: .center) {
                // Left: text info
                VStack(alignment: .leading, spacing: ForagerTheme.Spacing.xs) {
                    Text(weeklyList.name ?? "Unnamed List")
                        .font(ForagerTheme.cardTitle)
                        .foregroundStyle(isListCompleted ? ForagerTheme.textTertiary : ForagerTheme.textPrimary)

                    if let date = weeklyList.dateCreated {
                        Text(date, style: .date)
                            .font(ForagerTheme.captionFont)
                            .foregroundStyle(ForagerTheme.textTertiary)
                    }

                    Text("\(completedItemsCount) of \(totalItemsCount) items")
                        .font(ForagerTheme.secondaryFont)
                        .foregroundStyle(ForagerTheme.textSecondary)
                }

                Spacer()

                // Right: progress ring
                ForagerProgressRing(progress: completionPercentage)
            }

            // Divider
            Divider()
                .padding(.vertical, ForagerTheme.Spacing.sm)

            // Category chip pills
            CategoryChipPills(categories: categoryComposition)
        }
        .foragerCard()
    }

    // Computed: category breakdown for chip pills
    private var categoryComposition: [(name: String, count: Int)] {
        Dictionary(grouping: Array(itemsFetch)) { $0.categoryName ?? "Uncategorized" }
            .map { (name: $0.key, count: $0.value.count) }
            .sorted { $0.name < $1.name }
    }
}
```

### D.3: Auto-collapse fully completed categories

When all items in a category are checked, auto-collapse after 2s:

```swift
private func checkAutoCollapse(category: String, items: [GroceryListItem]) {
    let allCompleted = items.allSatisfy { $0.isCompleted }
    if allCompleted {
        DispatchQueue.main.asyncAfter(deadline: .now() + 2.0) {
            // Re-check: user may have unchecked an item during the 2s delay
            let stillAllCompleted = items.allSatisfy { $0.isCompleted }
            guard stillAllCompleted else { return }
            withAnimation(.spring(response: 0.3, dampingFraction: 0.8)) {
                collapsedCategories.insert(category)
            }
        }
    }
}

```swift
// In the view — haptics + animation only
private func toggleItemCompletion(_ item: GroceryListItem) {
    let wasCompleted = item.isCompleted

    // Haptic feedback
    if wasCompleted {
        UIImpactFeedbackGenerator(style: .light).impactOccurred()
    } else {
        UIImpactFeedbackGenerator(style: .medium).impactOccurred()
    }

    // Toggle with animation, persist via service
    withAnimation(.spring(response: 0.3, dampingFraction: 0.7)) {
        groceryListService.toggleCompletion(item)
    }
}

// In GroceryListService (or equivalent service that owns GroceryListItem writes)
func toggleCompletion(_ item: GroceryListItem) {
    item.isCompleted.toggle()
    item.dateCompleted = item.isCompleted ? Date() : nil
    do {
        try viewContext.save()
    } catch {
        handleError("Failed to toggle item completion", error: error)
    }
}
